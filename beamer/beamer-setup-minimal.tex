\usetheme{default}
\useoutertheme{default}
\usefonttheme[onlymath]{serif}
% define simplebeamercolor before \input to override
\providecolor{simplebeamercolor}{RGB}{57,89,199}
\usecolortheme[named=simplebeamercolor]{structure}

\setbeamertemplate{title page}{
    \vbox{}
    \vfill
    \noindent\hspace*{0.12\paperwidth}%
    \begin{minipage}{0.74\paperwidth}
        \raggedright
        {\usebeamerfont{title}\bfseries\color{black}\inserttitle\par}
        \ifx\insertsubtitle\empty\else
            \vspace{0.8em}
            {\usebeamerfont{subtitle}\color{black!70}\insertsubtitle\par}
        \fi
        \vspace{1.5em}
        {\color{simplebeamercolor}\rule{\linewidth}{0.8pt}\par}
        \vspace{2em}
        \ifx\insertauthor\empty\else
            {\usebeamerfont{author}\color{black}\insertauthor\par}
        \fi
        \ifx\insertinstitute\empty\else
            \vspace{0.5em}
            {\usebeamerfont{institute}\color{black!70}\insertinstitute\par}
        \fi
        \ifx\insertdate\empty\else
            \vspace{0.9em}
            {\usebeamerfont{date}\color{black}\insertdate\par}
        \fi
    \end{minipage}
    \vfill
}
\setbeamertemplate{section in toc}[sections numbered]
\setbeamertemplate{subsection in toc}[subsections numbered]
\addtobeamertemplate{subsection in toc}{\hspace{2em}}{}
\setbeamertemplate{itemize items}[circle]
\setbeamertemplate{enumerate items}[default]
\setbeamertemplate{caption}[numbered]
\setbeamertemplate{footline}[frame number]{}
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{frametitle}{\vspace*{0.5em}\insertframetitle\par\vskip-6pt\noindent\rule{\linewidth}{0.6pt}\vspace{-0.1em}}

\setbeamercolor{title}{use=structure,fg=white,bg=structure.fg}
\setbeamercolor{section in head/foot}{fg=white,bg=structure.fg}
\setbeamercolor{block body}{bg=structure!10}
\setbeamercolor{block title}{bg=structure!20}
\setbeamercolor{block body alerted}{bg=alerted text.fg!10}
\setbeamercolor{block title alerted}{bg=alerted text.fg!20}
\setbeamercolor{block body example}{bg=green!10}
\setbeamercolor{block title example}{bg=green!20}
